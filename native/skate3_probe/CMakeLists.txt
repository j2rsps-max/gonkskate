cmake_minimum_required(VERSION 3.25)
project(gonkskate_skate3_probe LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(gonkskate_skate3_probe STATIC src/probe.cpp)
target_include_directories(gonkskate_skate3_probe PUBLIC include)
target_compile_features(gonkskate_skate3_probe PUBLIC cxx_std_23)
target_link_libraries(gonkskate_skate3_probe PUBLIC Threads::Threads)
if(PROJECT_IS_TOP_LEVEL)
  enable_testing()
  add_executable(gonkskate-probe-test tests/probe_test.cpp)
  target_link_libraries(gonkskate-probe-test PRIVATE gonkskate_skate3_probe)
  add_test(NAME probe_contract COMMAND gonkskate-probe-test ${CMAKE_CURRENT_BINARY_DIR}/contract.jsonl)
endif()
