cmake_minimum_required(VERSION 3.20)
project(gonkskate_thug_parameters LANGUAGES CXX)
add_library(gonkskate_thug_parameters STATIC src/thug_params.cpp src/thug_flat_world.cpp src/thug_mesh_world.cpp)
target_include_directories(gonkskate_thug_parameters PUBLIC include)
target_compile_features(gonkskate_thug_parameters PUBLIC cxx_std_17)
include(CTest)
if(BUILD_TESTING)
    add_executable(gonkskate_thug_parameter_test tests/params_test.cpp)
    target_link_libraries(gonkskate_thug_parameter_test PRIVATE gonkskate_thug_parameters)
    add_test(NAME thug_parameter_semantics COMMAND gonkskate_thug_parameter_test)
    add_executable(gonkskate_thug_flat_world_test tests/flat_world_test.cpp)
    target_link_libraries(gonkskate_thug_flat_world_test PRIVATE gonkskate_thug_parameters)
    add_test(NAME thug_flat_world_queries COMMAND gonkskate_thug_flat_world_test)
    add_executable(gonkskate_thug_mesh_world_test tests/mesh_world_test.cpp)
    target_link_libraries(gonkskate_thug_mesh_world_test PRIVATE gonkskate_thug_parameters)
    add_test(NAME thug_mesh_world_queries COMMAND gonkskate_thug_mesh_world_test)
endif()
